# Trixor Logistics - Project Structure

```
trixor-logistics/
│
├── App.tsx                 # Главный компонент — собирает все секции сайта в одно приложение
├── index.tsx               # Точка входа — здесь React подключается к HTML через ReactDOM.render()
├── index.html              # HTML шаблон — базовый файл с <div id="root"> куда React вставляет контент
├── constants.ts            # Константы — тексты, цвета, настройки чтобы не хардкодить в компонентах
├── types.ts                # TypeScript типы — интерфейсы данных для типизации и автокомплита
├── tsconfig.json           # Конфиг TypeScript — правила компиляции, пути, версии
├── vite.config.ts          # Конфиг Vite — настройки сборщика, плагины, dev-сервер
├── package.json            # Манифест проекта — список зависимостей и npm скрипты (dev, build)
├── package-lock.json       # Лок зависимостей — фиксирует версии пакетов, НЕ РЕДАКТИРОВАТЬ
├── metadata.json           # Метаданные — информация о проекте
├── README.md               # Документация — описание проекта для GitHub
├── styles references.md    # Справочник стилей — гайд по дизайну и цветам
├── .env.local              # Секреты — API ключи, пароли (не попадает в Git)
├── .gitignore              # Git игнор — какие файлы не отслеживать (node_modules, .env)
│
├── components/                     # 📁 КОМПОНЕНТЫ — все UI блоки сайта
│   ├── Navbar.tsx                  #    Шапка сайта — лого, меню, кнопка "Request Rate"
│   ├── Hero.tsx                    #    Первый экран — видео фон, заголовок, CTA кнопки
│   ├── TrustStrip.tsx              #    Полоса доверия — цифры "10+ лет", "500+ клиентов"
│   ├── TrustMarquee.tsx            #    Бегущая строка — логотипы партнёров с анимацией
│   ├── Services.tsx                #    Услуги — карточки с описанием сервисов
│   ├── Equipment.tsx               #    Оборудование — типы грузовиков и прицепов
│   ├── Lanes.tsx                   #    Маршруты — карта направлений перевозок
│   ├── FleetEvidence.tsx           #    Автопарк — фотогалерея техники компании
│   ├── AboutSection.tsx            #    О компании — история, миссия, команда
│   ├── CTABanner.tsx               #    Баннер CTA — "Получите расчёт за 5 минут!"
│   ├── ContactForm.tsx             #    Форма контакта — поля ввода для заявки
│   ├── Vacancies.tsx               #    Вакансии — список открытых позиций
│   ├── Footer.tsx                  #    Подвал — контакты, ссылки, копирайт
│   └── ScrollReveal.tsx            #    Анимация — обёртка для fade-in эффекта при скролле
│
├── public/                         # 📁 СТАТИКА — файлы без обработки сборщиком
│   └── trixor/
│       ├── images/                 #    📁 Изображения
│       │   ├── Dispatcher1.png     #       Фото диспетчера для секции команды
│       │   ├── Driver1.png         #       Фото водителя #1
│       │   ├── Driver2.png         #       Фото водителя #2
│       │   ├── Mechanik2.png       #       Фото механика #2
│       │   ├── Mechnik1.png        #       Фото механика #1
│       │   ├── PHOTO-2025-...jpg   #       Дополнительное фото
│       │   ├── Trixor1.png         #       Лого или брендинг
│       │   └── TrixorWarehouse.png #       Фото склада/базы
│       │
│       └── video/                  #    📁 Видео
│           ├── Trixor hero v2.mp4  #       Фоновое видео для Hero (версия 2)
│           └── Trixor hero v3.mp4  #       Фоновое видео для Hero (версия 3)
│
└── node_modules/                   # 📁 ЗАВИСИМОСТИ — npm пакеты, НЕ ТРОГАТЬ
```
